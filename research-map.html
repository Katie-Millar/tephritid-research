    <button id="btn-cc"   style="padding:5px 14px;border-radius:4px;border:1px solid #C05C28;background:#fff;color:#C05C28;cursor:pointer;font-size:13px">🪰 <em>C. capitata</em></button>
    <span style="margin-left:auto;font-size:12px;color:#888;align-self:center">${institutions.length} institutions · click a marker for papers</span>
  </div>`
  yield bar

  // ── map ──
  const container = DOM.element("div", {
    style: "height:580px;width:100%;border-radius:8px;overflow:hidden;border:1px solid #ddd"
  })
  yield container

  const map = L.map(container, { center: [25, 20], zoom: 2, minZoom: 1 })
  L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
    attribution: "© OpenStreetMap contributors", maxZoom: 19
  }).addTo(map)

  const markers = []

  institutions.forEach(d => {
    const size = 28 + d.paperList.length * 4
    const icon = L.divIcon({
      html: `<div style="
        width:${size}px;height:${size}px;
        background:${colourFor(d.species)};
        border-radius:50%;border:2.5px solid #fff;
        box-shadow:0 2px 8px rgba(0,0,0,0.25);
        display:flex;align-items:center;justify-content:center;
        font-size:${size * 0.52}px;line-height:1">🪰</div>`,
      className: "", iconSize: [size, size],
      iconAnchor: [size/2, size/2], popupAnchor: [0, -(size/2)]
    })
    const m = L.marker([d.lat, d.lon], { icon })
      .bindPopup(popupFor(d), { maxWidth: 330 })
      .bindTooltip(`<strong>${d.name}</strong><br><em style="font-size:11px">${d.city}</em>`,
        { direction: "top", offset: [0, -(size/2)] })
    m._species = d.species
    m.addTo(map)
    markers.push(m)
  })

  // filter logic
  function setFilter(sp) {
    markers.forEach(m => {
      if (!sp || m._species === sp) m.addTo(map)
      else map.removeLayer(m)
    })
  }

  bar.querySelector("#btn-all").onclick = () => {
    setFilter(null)
    bar.querySelectorAll("button").forEach(b => {
      b.style.background = "#fff"; b.style.color = "#555"; b.style.borderColor = "#ccc"
    })
    const b = bar.querySelector("#btn-all")
    b.style.background = "#2E7D9E"; b.style.color = "#fff"; b.style.borderColor = "#2E7D9E"
  }
  bar.querySelector("#btn-bd").onclick = () => {
    setFilter("Bactrocera dorsalis")
    bar.querySelectorAll("button").forEach(b => {
      b.style.background = "#fff"; b.style.color = "#555"; b.style.borderColor = "#ccc"
    })
    const b = bar.querySelector("#btn-bd")
    b.style.background = "#2E7D9E"; b.style.color = "#fff"; b.style.borderColor = "#2E7D9E"
  }
  bar.querySelector("#btn-cc").onclick = () => {
    setFilter("Ceratitis capitata")
    bar.querySelectorAll("button").forEach(b => {
      b.style.background = "#fff"; b.style.color = "#555"; b.style.borderColor = "#ccc"
    })
    const b = bar.querySelector("#btn-cc")
    b.style.background = "#C05C28"; b.style.color = "#fff"; b.style.borderColor = "#C05C28"
  }

  // legend
  const legend = L.control({ position: "bottomleft" })
  legend.onAdd = () => {
    const div = L.DomUtil.create("div")
    div.innerHTML = `
      <div style="background:#fff;padding:10px 14px;border-radius:6px;
        border:1px solid #ccc;font-family:sans-serif;font-size:12px;line-height:1.9">
        <div style="font-weight:700;margin-bottom:4px">Species</div>
        <div><span style="display:inline-flex;align-items:center;justify-content:center;
          width:20px;height:20px;border-radius:50%;background:#2E7D9E;
          margin-right:6px;font-size:12px">🪰</span><em>B. dorsalis</em></div>
        <div><span style="display:inline-flex;align-items:center;justify-content:center;
          width:20px;height:20px;border-radius:50%;background:#C05C28;
          margin-right:6px;font-size:12px">🪰</span><em>C. capitata</em></div>
        <div style="margin-top:6px;color:#888;font-size:11px">Bubble size = paper count</div>
      </div>`
    return div
  }
  legend.addTo(map)
}
